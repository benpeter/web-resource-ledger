    ```js
    // tva
    // AUDIT-04: CDP-availability spike for Phase 7 (Subresource Experiment).
    //
    // This is a persistent integration test (NOT a one-shot script) so future
    // @cloudflare/playwright upgrades automatically re-validate the assumption
    // Phase 7 depends on. CDP support was announced for v1.3.0 (April 2026);
    // the project is currently pinned to ^1.1.2 -- this test is expected to
    // FAIL on the pinned version. Phase 1 will likely upgrade and re-run it.
    //
    // Exemption from the "log to Coralogix, never console" rule (CLAUDE.md
    // §Engineering Philosophy): this is test code, no `env` available; the
    // test runner captures stdout. The console.log emits the JSON evidence
    // line that AUDIT.md §3 cites verbatim.

    import { env } from 'cloudflare:test';
    import { describe, it, expect, beforeEach, inject } from 'vitest';
    import { acquire, connect } from '@cloudflare/playwright';

    async function ensureBrowserSession() {
      // miniflare browser binding doesn't implement limits(); pre-acquire to
      // warm the pool. Same pattern as capture-pipeline.test.js.
      const session = await acquire(env.BROWSER, { keep_alive: 120000 });
      const browser = await connect(env.BROWSER, session.sessionId);
      await browser.close();
    }

    describe('AUDIT-04 -- CDP availability on @cloudflare/playwright', () => {
      beforeEach(async () => {
        await ensureBrowserSession();
      });

      it('exposes page.context().newCDPSession(page) and Network.getResponseBody', async () => {
        const port = inject('testServerPort');
        const session = await acquire(env.BROWSER, { keep_alive: 60000 });
        const browser = await connect(env.BROWSER, session.sessionId);
        const ctx = await browser.newContext();
        const page = await ctx.newPage();

        let cdpSession;
        let cdpAvailable = false;
        let cdpError = null;
        try {
          expect(typeof page.context().newCDPSession).toBe('function');
          cdpSession = await page.context().newCDPSession(page);
          cdpAvailable = true;
        } catch (err) {
          cdpError = {
            name: err?.constructor?.name,
            message: String(err?.message ?? '').slice(0, 256),
          };
        }

        // Evidence line for AUDIT.md §3 (test runner captures stdout):
        // eslint-disable-next-line no-console
        console.log(JSON.stringify({
          test: 'AUDIT-04',
          cdpSessionAvailable: cdpAvailable,
          cdpError,
        }));

        if (!cdpAvailable) {
          await ctx.close();
          await browser.close();
          expect.fail(
            `newCDPSession unavailable on @cloudflare/playwright: ${JSON.stringify(cdpError)}`
          );
        }

        await cdpSession.send('Network.enable');

        const fixtureUrl = `http://127.0.0.1:${port}/fast.html`;
        const responsePromise = new Promise((resolve) => {
          cdpSession.on('Network.responseReceived', (params) => {
            if (params.response.url === fixtureUrl) resolve(params.requestId);
          });
        });

        await page.goto(fixtureUrl);
        const requestId = await responsePromise;

        const result = await cdpSession.send('Network.getResponseBody', { requestId });
        expect(typeof result.body).toBe('string');
        expect(result.body.length).toBeGreaterThan(0);

        await ctx.close();
        await browser.close();
      });
    });
    ```

    ```bash
    #!/bin/bash
    set -euo pipefail
    source ~/.secrets   # WRL_CAPTURE_API_KEY in scope

    while IFS= read -r URL; do
      [ -z "$URL" ] && continue
      [[ "$URL" == \#* ]] && continue
      echo "Capturing: $URL"
      RESPONSE=$(curl -s -X POST https://api.webresourceledger.com/v1/captures \
        -H "Authorization: Bearer $WRL_CAPTURE_API_KEY" \
        -H "Content-Type: application/json" \
        -d "{\"url\": \"$URL\"}")
      CAPTURE_ID=$(echo "$RESPONSE" | jq -r '.id // empty')
      if [ -z "$CAPTURE_ID" ]; then
        echo "ERROR: $URL -> $RESPONSE" >&2
        continue
      fi
      # Poll until terminal:
      for i in $(seq 1 60); do
        STATUS=$(curl -s "https://api.webresourceledger.com/v1/captures/$CAPTURE_ID" \
          -H "Authorization: Bearer $WRL_CAPTURE_API_KEY" | jq -r '.status')
        case "$STATUS" in
          complete|partial|failed) break;;
          *) sleep 5;;
        esac
      done
      printf '%s,%s,%s\n' "$URL" "$CAPTURE_ID" "$STATUS" >> .planning/audit/before-corpus.csv
    done < <(awk -F'|' '/^\|.*https?:/ {gsub(/[ \t]/,"",$3); print $3}' .planning/audit/url-battery.md)
    ```
